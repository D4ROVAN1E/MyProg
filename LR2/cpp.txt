#include <iostream> 
#include <string>  
using namespace std;

int main() {
    int N, M;        // Переменные для размеров доски (N - строки, M - столбцы)
    string S;        // Строка с командами
    cin >> N >> M >> S;

    // Проверка на некорректные размеры доски
    if (N < 1 || M < 1) {
        cout << "Invalid input" << endl; 
        return 0; 
    }

    // Проверка символов в строке команд на допустимость
    for (char c : S) {
        if (c != 'L' && c != 'R' && c != 'U' && c != 'D') {
            cout << "Invalid input" << endl; 
            return 0; 
        }
    }

    // Подсчёт количества команд каждого типа
    int L = 0, R = 0, U = 0, D = 0; // Счётчики для команд
    for (char c : S) {
        if (c == 'L') L++;      
        else if (c == 'R') R++;  
        else if (c == 'U') U++;  
        else if (c == 'D') D++;  
    }

    // Вычисление суммарного смещения по осям
    int delta_x = R - L; 
    int delta_y = D - U; 

    // Определение допустимых границ для начальной позиции по x
    int min_x = max(1, 1 - delta_x); 
    int max_x = min(M, M - delta_x);

    // Определение допустимых границ для начальной позиции по y
    int min_y = max(1, 1 - delta_y);
    int max_y = min(N, N - delta_y);

    // Проверка существования допустимой начальной позиции
    if (min_x > max_x || min_y > max_y) {
        cout << "No" << endl; 
    } else {
        // Вывод минимальных координат из допустимого диапазона (самая левая верхняя клетка)
        cout << "(" << min_x << "," << min_y << ")" << endl;
    }

    return 0; 
}
